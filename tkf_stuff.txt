(* 1. Core Parameters & Definitions *)
beta[t_, l_, m_] := (1 - Exp[(l - m) t]) / (m - l Exp[(l - m) t])
N0[t_, l_, m_] := m beta[t, l, m]
I1[t_, l_, m_] := 1 - l beta[t, l, m]
H1[t_, l_, m_] := Exp[-m t] * (1 - l beta[t, l, m])
U[t_, l_, m_] := 1 - Exp[-m t] - m beta[t, l, m]

(* 2. Global Symbolic Simplifications for Stability *)
constraints = {l > 0, m > 0, l < m, t > 0};

symbolicBeta = FullSimplify[beta[t, l, m], constraints];
symbolicLogBeta = FullSimplify[Log[symbolicBeta], constraints];

symbolicI1 = FullSimplify[I1[t, l, m], constraints];
symbolicLogI1 = FullSimplify[Log[symbolicI1], constraints];

symbolicH1 = FullSimplify[H1[t, l, m], constraints];
symbolicLogH1 = FullSimplify[Log[symbolicH1], constraints];

symbolicU = FullSimplify[U[t, l, m], constraints];
symbolicLogU = FullSimplify[Log[symbolicU], constraints];

Print["--- Symbolic Analysis ---"];
Print["Simplified beta: ", symbolicBeta];
Print["Simplified I1: ", symbolicI1];
Print["Simplified H1: ", symbolicH1];
Print["Simplified U: ", symbolicU];
Print["\n"];

Print["--- Symbolic Log Analysis ---"];
Print["Simplified Log[beta]: ", symbolicLogBeta];
Print["Simplified Log[I1]: ", symbolicLogI1];
Print["Simplified Log[H1]: ", symbolicLogH1];
Print["Simplified Log[U]: ", symbolicLogU];
Print["\n"];

(* Numerically stable symbolic form of Log[U] to avoid catastrophic cancellation (1 - approx(1)) *)
(* This was found by prompting opencode to make the FullSimplify of Log[U] = Log[1 - e^{-mt} - m \frac{1 - e^{(l-m)t}}{m - l e^{(l-m)t}}] more robust *)
symbolicLogUStable = (l - m) t + Log[(m - l) - m*Exp[-l*t] + l*Exp[-m*t]] - Log[m - l*Exp[(l - m) t]];

(* Analytical proof that symbolicLogU == symbolicLogUStable *)
Print["--- Analytical Proof ---"];
Print["Log[U] simplified: ", symbolicLogU];
Print["Log[U] stable form: ", symbolicLogUStable];
proof = FullSimplify[Exp[symbolicLogU - symbolicLogUStable], constraints];
Print["Exp[Log[U] - Log[U]stable] = ", proof];
Print["The expressions are analytically equal: ", proof === 1];
Print["\n"];

Print["--- Taylor ---"];
(* Normal converts a series expansion (returned by Series) into a regular polynomial/expression by dropping the O[t] remainder term. *)
Print["Taylor expansion of ln(beta(t)) at t=0: ", Normal[Series[Log[beta[t, l, m]], {t, 0, 3}]]];

(* 
   safeLogN[expr, tVal, lVal, mVal] is a wrapper to handle the transition from 
   symbolic formulas to high-precision numerical values. Does not approximate except 
   for returning the result.

   1. It converts the inputs to their exact binary values via SetPrecision[.., Infinity].
      (Rationalize[tVal, 0] must NOT be used here: it maps numbers that differ only in
      their last bits -- e.g. two adjacent doubles for lambda and mu -- to the same
      rational, which can turn beta and therefore ln(beta) into 0/0 and Indeterminate.)
   2. It substitutes these exact values into the pre-simplified symbolic 'expr'.
   3. It uses Block to temporarily allow up to 1000 digits of internal 
      $MaxExtraPrecision for sensitive log calculations.
   4. It returns Indeterminate if the calculation fails, hits numerical limits,
      or comes back with fewer than 900 of the requested 1000 digits of precision.
      The precision check is essential: cancellation can silently consume digits,
      and a finite but low-precision result would otherwise pass through and be
      mistaken for a converged value.
*)
safeLogN[expr_, tVal_, lVal_, mVal_] := Module[{exactExpr, val, T, L, M},
    T = SetPrecision[tVal, Infinity];
    L = SetPrecision[lVal, Infinity];
    M = SetPrecision[mVal, Infinity];
    exactExpr = expr /. {t -> T, l -> L, m -> M};
    
    val = Block[{$MaxExtraPrecision = 1000}, N[exactExpr, 1000]];
    
    If[Not[NumberQ[val]] || Abs[val] == Infinity || Precision[val] < 900,
        Indeterminate,
        val
    ]
]

logBetaSym[tVal_, lVal_, mVal_] := safeLogN[symbolicLogBeta, tVal, lVal, mVal]
logH1Sym[tVal_, lVal_, mVal_] := safeLogN[symbolicLogH1, tVal, lVal, mVal]
logI1Sym[tVal_, lVal_, mVal_] := safeLogN[symbolicLogI1, tVal, lVal, mVal]

(* Exact high-precision logs of the inputs themselves. Log[machineNumber] would
   only carry machine precision and would truncate the surrounding 1000-digit sum. *)
logMSym[mVal_] := Block[{$MaxExtraPrecision = 1000}, N[Log[SetPrecision[mVal, Infinity]], 1000]]
logLSym[lVal_] := Block[{$MaxExtraPrecision = 1000}, N[Log[SetPrecision[lVal, Infinity]], 1000]]

logUSym[tVal_, lVal_, mVal_] := Module[{res, stableVal},
    res = Quiet[safeLogN[symbolicLogU, tVal, lVal, mVal]];
    stableVal = safeLogN[symbolicLogUStable, tVal, lVal, mVal];
    
    If[res =!= Indeterminate,
        If[res =!= stableVal, 
            Print["Warning: Significant difference between raw and stable Log[U] at t=", tVal, " (", res, " vs ", stableVal, ")"]
        ];,
        Print["Warning: Raw Log[U] is Indeterminate,  using stable version instead."];
    ];
    
    stableVal
]

logN0Sym[tVal_, lVal_, mVal_] := logBetaSym[tVal, lVal, mVal] + logMSym[mVal]
Eta[t_, l_, m_] := logUSym[t, l, m] + logI1Sym[t, l, m] - logMSym[m] - logLSym[l] - 2* logBetaSym[t, l, m]

(* One flat list of {lambda, mu, time} triples used by every table.
   Join, not DeleteDuplicates: SameQ treats adjacent machine doubles as equal,
   which would silently drop the one-ulp boundary cases above. *)
allCases = Join[paramTimeCases, extraCases];

(* Table Utility *)
formatVal[v_, prec_: 16] := Module[{vN, mantissa, exponent},
    If[Not[NumberQ[v]], Return[ToString[v]]];
    (* SetPrecision (not N) so that machine numbers can be displayed with more
       than their built-in ~16 digits; needed to distinguish lambda and mu that
       differ by one ulp. High-precision inputs are unaffected. *)
    vN = SetPrecision[v, prec + 4];
    If[Not[NumberQ[vN]], Return[ToString[vN]]];
    If[vN == 0, Return[ToString[NumberForm[0, {prec + 4, prec}]]]];
    If[Abs[vN] < 10^-3 || Abs[vN] > 10^5,
        (* Normalize to 1 <= |mantissa| < 10; MantissaExponent uses a different
           mantissa range for machine numbers and for arbitrary-precision numbers. *)
        {mantissa, exponent} = MantissaExponent[vN];
        While[Abs[mantissa] >= 10, mantissa = mantissa/10; exponent = exponent + 1];
        While[Abs[mantissa] < 1, mantissa = mantissa*10; exponent = exponent - 1];
        ToString[NumberForm[mantissa, {prec + 4, prec}]] <> "e" <> ToString[exponent],
        ToString[NumberForm[vN, {prec + 4, prec}]]
    ]
]

printTable[title_, func_, cases_] := Module[{valueDigits = 16},
    Print["--- ", title, " ---"];
    Print["| Time          | Lambda                     | Mu                         | Value                  | Warn   |"];
    Print["|---------------|----------------------------|----------------------------|------------------------|--------|"];
    Do[
        lVal = c[[1]]; mVal = c[[2]]; tVal = c[[3]];
        val = func[tVal, lVal, mVal];
        tStr = StringPadRight[formatVal[tVal, 4], 13];
        (* 18 decimal places: enough to distinguish one-ulp-separated
           lambda/mu even for values near 1e-2. *)
        lStr = StringPadRight[formatVal[lVal, 18], 26];
        mStr = StringPadRight[formatVal[mVal, 18], 26];
        valStr = StringPadRight[formatVal[val, valueDigits], 24];
        (* Nonzero: warn on relative error (precision).
           val == 0 (including underflow): relative error is undefined, so
           instead require absolute error < 10^-valueDigits, i.e.
           Accuracy[val] > valueDigits. *)
        warn = Which[
            !NumberQ[val], "BAD",
            val == 0 && Accuracy[val] <= valueDigits, "LOW-A",
            val != 0 && Precision[val] < valueDigits, "LOW-P",
            True, ""
        ];
        Print["| ", tStr, " | ", lStr, " | ", mStr, " | ", valStr,
              " | ", StringPadRight[warn, 6], " |"];
    , {c, cases}];
    Print["\n"];
]

printTable["ln(beta)", logBetaSym, allCases];

printTable["ln(N0)", logN0Sym, allCases];
printTable["ln(U)", logUSym, allCases];
(* eta is also in log space *)
printTable["Eta", Eta, allCases];
